"Clean up a book title stored in 'bookTitle' (e.g., "the_hunger_games"):"
string bookTitle = "the_hunger_games";
bookTitle = bookTitle.Replace('_', ' ');
string[] bookParts = bookTitle.Split(' ');
for(int i = 0; i < bookParts.Length; i++){
    bookParts[i] = bookParts[i].Substring(0, 1).ToUpper() + bookParts[i].Substring(1).ToLower();
}
string cleanTitle = String.Join(" ", bookParts);
// Your code here
return cleanTitle;
